rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper function to get user role (with null check)
    function getUserRole() {
      let userDoc = get(/databases/$(database)/documents/users/$(request.auth.uid));
      return userDoc != null ? userDoc.data.role : null;
    }
    
    function isOwner() {
      return request.auth != null && getUserRole() == 'owner';
    }
    
    function isAdminOrOwner() {
      let role = getUserRole();
      return request.auth != null && role in ['admin', 'owner'];
    }
    
    // Helper function to check if current user is God Owner
    function isGodOwner() {
      let userDoc = get(/databases/$(database)/documents/users/$(request.auth.uid));
      return userDoc != null && userDoc.data.isGodOwner == true;
    }
    
    // Users collection
    match /users/{userId} {
      
      // Users can read their own data, owners can read all
      allow read: if request.auth != null && 
        (request.auth.uid == userId || isOwner());
      
      // Users can create their own document on signup
      allow create: if request.auth != null && 
        (request.auth.uid == userId);
      
      // Users can update their own documents (for registeredEvents and savedEvents)
      // Only God Owner can update user roles to owner or transfer God Owner status
      // Regular owners can update roles for admin approval
      // UPDATED: God Owner system - Phase 0.5
      // UPDATED: Fixed user document update permissions for registration/saving
      allow update: if request.auth != null && 
        (isGodOwner() ||
         (isOwner() && request.resource.data.isGodOwner == resource.data.isGodOwner) ||
         (request.auth.uid == userId && 
          request.resource.data.role == resource.data.role &&
          request.resource.data.isGodOwner != true));
      allow delete: if request.auth != null && isOwner();
    }
    
    // Events collection
    match /events/{eventId} {
      // Public can read approved events only
      // Owners and admins can read all events (including pendingApproval, draft, rejected)
      // UPDATED: role-based routing and approval flows - Phase 0.5
      allow read: if resource == null || 
        resource.data.status == 'approved' || 
        (request.auth != null && isAdminOrOwner());
      
      // Owners and admins can create events
      allow create: if request.auth != null && isAdminOrOwner();
      
      // Owners can update any event
      // Admins can only update their own events
      // Authenticated users can update registeredUsers and savedByUsers arrays (for registration/saving)
      // UPDATED: Fixed registration/save permissions - simplified to allow array updates
      allow update: if request.auth != null && 
        (isOwner() || 
         (getUserRole() == 'admin' && resource.data.createdBy == request.auth.uid) ||
         // Allow any authenticated user to update registeredUsers and savedByUsers arrays
         // Only verify that immutable required fields haven't changed
         // Note: updateDoc() only updates specified fields, so other fields remain unchanged
         (request.resource.data.title == resource.data.title &&
          request.resource.data.date == resource.data.date &&
          request.resource.data.location == resource.data.location &&
          request.resource.data.price == resource.data.price &&
          request.resource.data.status == resource.data.status &&
          request.resource.data.category == resource.data.category &&
          request.resource.data.createdBy == resource.data.createdBy &&
          request.resource.data.createdByEmail == resource.data.createdByEmail));
      
      // Owners can delete any event
      // Admins can only delete their own events
      allow delete: if request.auth != null && 
        (isOwner() || 
         (getUserRole() == 'admin' && resource.data.createdBy == request.auth.uid));
    }
    
    // Admin Requests collection
    // UPDATED: role-based routing and approval flows - Phase 0.5
    match /adminRequests/{requestId} {
      // Players can read their own pending requests
      // Owners can read all requests
      allow read: if request.auth != null && 
        (resource.data.userId == request.auth.uid || isOwner());
      
      // Players can create their own admin requests
      allow create: if request.auth != null && 
        request.resource.data.userId == request.auth.uid &&
        request.resource.data.status == 'pending';
      
      // Only owners can update (approve/reject) admin requests
      allow update: if request.auth != null && isOwner();
      
      // Only owners can delete admin requests
      allow delete: if request.auth != null && isOwner();
    }
  }
}
